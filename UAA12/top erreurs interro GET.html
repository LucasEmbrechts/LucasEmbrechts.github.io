Échange de données via GET

Top 10 des (bêtes) erreurs

❌ ERREUR #1 : Noms des clés différents

JS envoie "q1" mais PHP cherche "question1"

// JavaScript
let q1 = document.getElementById("q1").value;
location.href = "quiz.php?q1=" + q1;

// PHP
if(isset($_GET["question1"])) {
$reponse = $_GET["question1"];
}
❌ q1 ≠ question1
⚠️ $_GET["question1"] n'existe jamais
✅ SOLUTION #1 : MÊMES noms partout

Cohérence JavaScript ↔ PHP

// JavaScript
let q1 = document.getElementById("q1").value;
location.href = "quiz.php?q1=" + q1;

// PHP
if(isset($_GET["q1"])) {
$reponse = $_GET["q1"];
}
✅ Même nom "q1" partout
✅ Données transmises correctement
❌ ERREUR #2 : URL malformée

Manque & ou = dans l'URL

// INCORRECT - manque &
location.href = "quiz.php?q1=" + q1 + "q2=" + q2;

// INCORRECT - manque =
location.href = "quiz.php?q1" + q1 + "&q2" + q2;
❌ Sans & et = l'URL est invalide
⚠️ PHP ne peut pas lire les données
✅ SOLUTION #2 : & entre paramètres

Structure : ?param1=val&param2=val

// CORRECT
location.href = "quiz.php?q1=" + q1 + "&q2=" + q2 + "&q3=" + q3;
✅ Chaque paramètre séparé par &
✅ Toutes les données transmises
❌ ERREUR #3 : Pas de isset()

Accéder à $_GET sans vérifier

// INCORRECT
$q1 = $_GET["q1"];
$q2 = $_GET["q2"];

if($q1 == "$") {
$score++;
}
❌ Si paramètres absents = erreur
⚠️ "Undefined array key" - crash PHP
✅ SOLUTION #3 : TOUJOURS isset()

Vérifier avant d'utiliser $_GET

// CORRECT
if(isset($_GET["q1"]) && isset($_GET["q2"])) {
$q1 = $_GET["q1"];
$q2 = $_GET["q2"];
if($q1 == "$") {
$score++;
}
}
✅ isset() vérifie l'existence
✅ Plus d'erreur lorsque l'utilisateur arrive la première fois sur la page
❌ ERREUR #4 : Oublier .value

Récupérer l'élément au lieu de la valeur

// INCORRECT
let q1 = document.getElementById("q1");
let cb = document.getElementById("checkbox");

location.href = "quiz.php?q1=" + q1 + "&montrer=" + cb;
❌ q1 = élément HTML, pas le texte
⚠️ URL = "[object HTMLInputElement]"
✅ SOLUTION #4 : .value et .checked

Récupérer la vraie valeur

// CORRECT
let q1 = document.getElementById("q1").value;
let cb = document.getElementById("checkbox").checked;

location.href = "quiz.php?q1=" + q1 + "&montrer=" + cb;
✅ .value = texte, .checked = true/false
✅ Vraies valeurs transmises
❌ ERREUR #5 : Pas de location.href

Fonction qui ne redirige jamais

// INCORRECT
function envoyer() {
let q1 = document.getElementById("q1").value;
let url = "quiz.php?q1=" + q1;
// RIEN !
}
❌ URL créée mais jamais utilisée
⚠️ Rien ne se passe au clic
✅ SOLUTION #5 : location.href

Rediriger vers la page PHP

// CORRECT
function envoyer() {
let q1 = document.getElementById("q1").value;
location.href = "quiz.php?q1=" + q1;
}
✅ location.href charge la nouvelle URL
✅ Page rechargée avec données GET
❌ ERREUR #6 : $_GET sans guillemets

Oublier les guillemets sur la clé

// INCORRECT
$q1 = $_GET[question1];
$q2 = $_GET[question2];
❌ Clés doivent être entre guillemets
⚠️ "Use of undefined constant"
✅ SOLUTION #6 : Guillemets obligatoires

Syntaxe correcte pour $_GET

// CORRECT
$q1 = $_GET["question1"];
$q2 = $_GET["question2"];

// ou avec guillemets simples
$q1 = $_GET['question1'];

✅ Clés = chaînes → entre guillemets
✅ Syntaxe PHP valide
❌ ERREUR #7 : Checkbox sans name

Seulement un id, pas de name

<!-- INCORRECT -->
<input type="checkbox" id="montrer">
❌ Sans name, pas envoyé en GET
⚠️ $_GET["montrer"] n'existe jamais
✅ SOLUTION #7 : name ET id

name obligatoire pour GET

<!-- CORRECT -->
<input type="checkbox" id="montrer" name="montrer">

<!-- id = JS, name = GET -->
✅ name définit le paramètre GET
✅ Paramètre récupérable en PHP
❌ ERREUR #8 : Checkbox == true

Comparer avec booléen en PHP

// INCORRECT
if($_GET["montrer"] == true) {
// afficher correctif
}

$montrer = $_GET["montrer"];
if($montrer == true) {
// ...
}
❌ JS envoie "true"/"false" (chaînes)
⚠️ Condition peut être fausse
✅ SOLUTION #8 : Comparer avec "true"

JavaScript envoie des chaînes

// CORRECT
if($_GET["montrer"] == "true") {
// afficher correctif
}

// ou mieux
if(isset($_GET["montrer"]) && $_GET["montrer"] == "true") {
// ...
}
✅ Comparer avec chaîne "true"
✅ Condition fonctionne
❌ ERREUR #9 : Score jamais incrémenté

Oublier $score++

// INCORRECT
$score = 0;

if($q1 == "$") {
echo "Correct !";
}

echo "Score : " . $score . "/5";
❌ Score reste à 0
⚠️ Affiche toujours 0/5
✅ SOLUTION #9 : $score++

Incrémenter à chaque bonne réponse

// CORRECT
$score = 0;

if($q1 == "$") {
echo "Correct !";
$score++;
}

echo "Score : " . $score . "/5";
✅ $score++ augmente de 1
✅ Score correct affiché
❌ ERREUR #10 : OR au lieu de AND

Utiliser || au lieu de &&

// INCORRECT
if($score >= 0 || $score <= 2) {
echo "Continue";
}

if(isset($_GET["q1"]) || isset($_GET["q2"])) {
// Un seul suffit
}
❌ || = vrai si UNE condition vraie
⚠️ Toujours vrai ou trop tôt
✅ SOLUTION #10 : AND (&&)

&& exige TOUTES les conditions

// CORRECT
if($score >= 0 && $score <= 2) {
echo "Continue";
}

if(isset($_GET["q1"]) && isset($_GET["q2"])) {
// TOUS doivent exister
}
✅ && = vrai si TOUTES vraies
✅ Logique correcte
